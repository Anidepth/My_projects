sequenceDiagram actor User participant Browser participant login.php participant config.php participant MySQL participant Session User->>Browser: Open index.html Browser->>login.php: Click Login link login.php-->>Browser: Render login form User->>Browser: Enter username, password, role Browser->>login.php: POST username + password + role login.php->>config.php: include config.php config.php->>MySQL: new mysqli(localhost, root, fish_accessories_store) MySQL-->>config.php: $conn established config.php-->>login.php: $conn ready login.php->>MySQL: SELECT id, username, password, role FROM users WHERE username=? AND role=? MySQL-->>login.php: Result row alt User found login.php->>login.php: password_verify($password, $hash) alt Password correct login.php->>Session: $_SESSION[user_id, username, role] alt role = admin login.php-->>Browser: Redirect → admin-dashboard.php else role = user login.php-->>Browser: Redirect → user-dashboard.php end else Password wrong login.php-->>Browser: Show "Invalid password!" error end else User not found login.php-->>Browser: Show "User not found!" error end User->>Browser: Click Logout Browser->>logout.php: GET logout.php logout.php->>Session: session_destroy() logout.php-->>Browser: Redirect → index.html
